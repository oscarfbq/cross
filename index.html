// --- NUEVA MATEMÁTICA: Piezas[i-1] + Metros[i] = Piezas[i] --- function realizarCalculos() { const rows = document.querySelectorAll('#table-body tr'); let perfilActual = ""; let metaActual = 0; let acumuladoOk = 0; let acumuladoNoOk = 0; let lastScrapAcumText = ""; let lastScrapAcumColor = ""; let propagarCascada = false; rows.forEach((row, index) => { const selectVal = row.querySelector('.perfil-input').value; const inputMetaStr = row.querySelector('.total-input').value; const inputOkM = row.querySelector('.ok-m-input'); const inputNoOkM = row.querySelector('.nook-m-input'); const inputOkP = row.querySelector('.ok-p-input'); const inputNoOkP = row.querySelector('.nook-p-input'); const scrapInput = row.querySelector('.scrap-input'); const scrapAcumInput = row.querySelector('.scrap-acum-input'); let topOkStr = inputOkM.value.trim(); let topNoOkStr = inputNoOkM.value.trim(); let bottomOkStr = inputOkP.value.trim(); let bottomNoOkStr = inputNoOkP.value.trim(); let isFirstRowOfSequence = false; // Detecta si es la primera hora del turno o si cambiaste de perfil if (selectVal !== "" && selectVal !== perfilActual) { perfilActual = selectVal; propagarCascada = true; isFirstRowOfSequence = true; } else if (index === 0) { propagarCascada = true; isFirstRowOfSequence = true; } const inputMeta = parseFloat(inputMetaStr); if (!isNaN(inputMeta) && inputMeta > 0) { metaActual = inputMeta; } // ------------------------------------------------------------- // NUEVA LÓGICA: AUTO-CALCULAR METROS NO OK // ------------------------------------------------------------- let topOk = parseFloat(topOkStr); if (metaActual > 0 && !isNaN(topOk) && topOkStr !== "") { // Calculamos la diferencia let diferencia = metaActual - topOk; // Asignamos el valor al input de Metros NoOk inputNoOkM.value = diferencia; // Actualizamos el string para que el resto del código lo procese topNoOkStr = diferencia.toString(); } topOk = topOk || 0; let topNoOk = parseFloat(topNoOkStr) || 0; // ------------------------------------------------------------- let bottomOk = parseFloat(bottomOkStr) || 0; let bottomNoOk = parseFloat(bottomNoOkStr) || 0; let hasHourlyData = topOkStr !== "" || topNoOkStr !== "" || topOkStr === "0" || topNoOkStr === "0" || bottomOkStr !== "" || bottomNoOkStr !== ""; if (isFirstRowOfSequence) { acumuladoOk = bottomOk; acumuladoNoOk = bottomNoOk; if (hasHourlyData) { if (metaActual > 0) { let scrapFila = (topNoOk / metaActual) * 100; scrapInput.value = parseFloat(scrapFila.toFixed(1)) + '%'; scrapInput.style.color = scrapFila > 5 ? '#e53935' : '#00a650'; } else if (topNoOk > 0) { scrapInput.value = 'Falta Meta'; scrapInput.style.color = '#e53935'; } else { scrapInput.value = ''; } let granTotal = acumuladoOk + acumuladoNoOk; if (granTotal > 0) { let scrapAcum = (acumuladoNoOk / granTotal) * 100; lastScrapAcumText = parseFloat(scrapAcum.toFixed(1)) + '%'; lastScrapAcumColor = scrapAcum > 5 ? '#e53935' : '#00a650'; scrapAcumInput.value = lastScrapAcumText; scrapAcumInput.style.color = lastScrapAcumColor; } else { scrapAcumInput.value = '0%'; scrapAcumInput.style.color = '#00a650'; lastScrapAcumText = '0%'; lastScrapAcumColor = '#00a650'; } } else { scrapInput.value = ''; scrapAcumInput.value = ''; } propagarCascada = true; } else if (propagarCascada) { if (topOkStr !== "" || topNoOkStr !== "" || topOkStr === "0" || topNoOkStr === "0") { acumuladoOk = acumuladoOk + topOk; acumuladoNoOk = acumuladoNoOk + topNoOk; inputOkP.value = acumuladoOk; inputNoOkP.value = acumuladoNoOk; if (metaActual > 0) { let scrapFila = (topNoOk / metaActual) * 100; scrapInput.value = parseFloat(scrapFila.toFixed(1)) + '%'; scrapInput.style.color = scrapFila > 5 ? '#e53935' : '#00a650'; } else if (topNoOk > 0) { scrapInput.value = 'Falta Meta'; scrapInput.style.color = '#e53935'; } else { scrapInput.value = ''; } let granTotal = acumuladoOk + acumuladoNoOk; if (granTotal > 0) { let scrapAcum = (acumuladoNoOk / granTotal) * 100; lastScrapAcumText = parseFloat(scrapAcum.toFixed(1)) + '%'; lastScrapAcumColor = scrapAcum > 5 ? '#e53935' : '#00a650'; scrapAcumInput.value = lastScrapAcumText; scrapAcumInput.style.color = lastScrapAcumColor; } else { scrapAcumInput.value = '0%'; scrapAcumInput.style.color = '#00a650'; lastScrapAcumText = '0%'; lastScrapAcumColor = '#00a650'; } } else { inputOkP.value = ""; inputNoOkP.value = ""; scrapInput.value = ''; if (metaActual > 0 && lastScrapAcumText !== "") { scrapAcumInput.value = lastScrapAcumText; scrapAcumInput.style.color = lastScrapAcumColor; } else { scrapAcumInput.value = ''; } propagarCascada = false; } } else { inputOkP.value = ""; inputNoOkP.value = ""; scrapInput.value = ''; scrapAcumInput.value = ''; } }); }
